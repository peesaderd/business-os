  <script>
    // ── State ──
    const state = {
      provider: null,
      category: null,
      product: null,
      variant: null,
      artworkUrl: null,
      mockupUrl: null,
      mockupId: null,
      products: [],
      variants: []
    };

    let steps = ['provider', 'category', 'product', 'variant', 'artwork', 'result'];

    // ── Toast ──
    let toastTimer;

    function toast(msg, type = 'info') {
      const el = document.getElementById('toast');
      el.textContent = msg;
      el.className = 'toast ' + type + ' show';
      clearTimeout(toastTimer);
      toastTimer = setTimeout(() => el.classList.remove('show'), 3000);
    }

    // ── Status ──
    function setStatus(text, type = 'idle') {
      document.getElementById('statusText').textContent = text;
      document.getElementById('statusDot').className = 'status-dot ' + type;
    }

    function setProgress(pct) {
      document.getElementById('progressFill').style.width = pct + '%';
    }

    // ── Step Navigation ──
    function showStep(name) {
      document.querySelectorAll('.card').forEach(c => c.classList.add('hidden'));
      const el = document.getElementById('step-' + name);
      if (el) el.classList.remove('hidden');

      document.querySelectorAll('.step').forEach((s, i) => {
        s.classList.remove('active', 'completed');
        if (steps[i] === name) s.classList.add('active');
        const idx = steps.indexOf(name);
        if (idx > 0 && i < idx) s.classList.add('completed');
      });
    }

    function resetWizard() {
      state.provider = null; state.category = null; state.product = null;
      state.variant = null; state.artworkUrl = null; state.mockupUrl = null;
      document.getElementById('artworkUrl').value = '';
      showStep('provider');
      setStatus('🟢 พร้อมทำงาน — เลือก Provider เพื่อเริ่มต้น', 'idle');
      setProgress(0);
    }

    // ── Provider ──
    function selectProvider(name) {
      state.provider = name;
      toast('✅ เลือก ' + name, 'success');
      setStatus('📂 กำลังโหลดหมวดหมู่...', 'loading');
      setProgress(16);
      setTimeout(() => loadCategories(name), 100);
    }

    // ── Categories ──
    async function loadCategories(provider) {
      try {
        let cats = [];

        if (provider === 'printful') {
          const r = await fetch('/api/pod/categories');
          const data = await r.json();
          cats = (data.result || []).map(c => ({
            id: c.id,
            name: c.name,
            icon: c.icon || '📦'
          }));
        }

        const list = document.getElementById('categoryList');
        list.innerHTML = cats.map(c =>
          `<button class="btn btn-secondary btn-block" onclick="selectCategory('${c.id}','${c.name}')" style="padding:16px;flex-direction:column;gap:8px;">
            <span style="font-size:28px;">${c.icon}</span>
            <span>${c.name}</span>
          </button>`
        ).join('') || '<div style="color:var(--text-dim);text-align:center;grid-column:1/-1;padding:24px;">⚠️ ไม่พบหมวดหมู่</div>';

        showStep('category');
        setStatus('📂 เลือกหมวดหมู่สินค้า', 'idle');
        setProgress(16);
      } catch (e) {
        toast('❌ โหลดหมวดหมู่ไม่สำเร็จ: ' + e.message, 'error');
        setStatus('🔴 Error: ' + e.message, 'error');
      }
    }

    function selectCategory(id, name) {
      state.category = { id, name };
      toast('✅ เลือก: ' + name, 'success');
      setStatus('👕 กำลังโหลดสินค้า...', 'loading');
      setProgress(33);
      loadProducts();
    }

    // ── Products ──
    async function loadProducts() {
      try {
        const r = await fetch('/api/pod/products?category_id=' + state.category.id);
        const data = await r.json();
        state.products = (data.result || []).map(p => ({
          id: p.id,
          name: p.name || p.product?.name || 'Product #' + p.id,
          image: p.image || p.product?.image || '',
          variants: (p.variants || []).length
        }));

        showStep('product');
        renderProducts();
        setStatus('👕 เลือกสินค้าที่ต้องการ', 'idle');
        setProgress(33);
      } catch (e) {
        toast('❌ โหลดสินค้าไม่สำเร็จ', 'error');
        setStatus('🔴 Error: ' + e.message, 'error');
      }
    }

    function renderProducts() {
      const list = document.getElementById('productList');
      list.innerHTML = state.products.map(p =>
        `<div class="card" style="cursor:pointer;padding:16px;" onclick="selectProduct('${p.id}')">
          <div style="display:flex;align-items:center;gap:16px;">

    }

    function filterProducts() {
      const q = document.getElementById('productSearch').value.toLowerCase();
      const filtered = state.products.filter(p => p.name.toLowerCase().includes(q));
      const list = document.getElementById('productList');
      list.innerHTML = filtered.map(p =>
        `<div class="card" style="cursor:pointer;padding:16px;" onclick="selectProduct('${p.id}')">
          <div style="display:flex;align-items:center;gap:16px;">
            ${p.image ? `<img src="${p.image}" alt="" style="width:64px;height:64px;object-fit:contain;border:var(--pixel-size) solid var(--border-color);" />` : '<div style="width:64px;height:64px;background:var(--bg);display:flex;align-items:center;justify-content:center;font-size:28px;">👕</div>'}
            <div>
              <strong style="font-size:15px;">${p.name}</strong>
              <div style="font-size:12px;color:var(--text-dim);">${p.variants} variants</div>
            </div>
          </div>
        </div>`
      ).join('');
    }

    function selectProduct(id) {
      state.product = state.products.find(p => String(p.id) === String(id));
      toast('✅ เลือก: ' + state.product.name, 'success');
      setStatus('🎨 กำลังโหลด Variant...', 'loading');
      setProgress(50);
      loadVariants(id);
    }

    // ── Variants ──
    async function loadVariants(productId) {
      try {
        const r = await fetch('/api/pod/products/' + productId);
        const data = await r.json();
        const p = data.result?.product || data.result || {};
        state.variants = (p.variants || []).map(v => ({
          id: v.id,
          name: v.name || v.size + (v.color ? ' — ' + v.color : ''),
          size: v.size || '',
          color: v.color || ''
        }));

        const list = document.getElementById('variantList');
        list.innerHTML = state.variants.map(v =>
          `<button class="btn btn-secondary btn-block" onclick="selectVariant('${v.id}')" style="padding:16px;flex-direction:column;gap:4px;">
            <span style="font-size:14px;font-weight:600;">${v.name}</span>
            ${v.size ? `<span style="font-size:11px;color:var(--text-dim);">${v.size}</span>` : ''}
          </button>`
        ).join('');

        showStep('variant');
        setStatus('🎨 เลือก Variant (ขนาด/สี)', 'idle');
        setProgress(50);
      } catch (e) {
        toast('❌ โหลด Variant ไม่สำเร็จ', 'error');
        setStatus('🔴 Error: ' + e.message, 'error');
      }
    }

    function selectVariant(id) {
      state.variant = state.variants.find(v => String(v.id) === String(id));
      toast('✅ เลือก: ' + state.variant.name, 'success');
      state.designUrl = '';
      state.artworkUrl = '';
      showStep('artwork');
      setStatus('🎨 กำลังสร้างลายด้วย AI...', 'loading');
      setProgress(66);
      // Auto-generate AI design
      generateDesign();
    }

    // ── AI Design Generation ──
    async function generateDesign() {
      const btn = document.getElementById('genDesignBtn');
      const placeholder = document.getElementById('designPlaceholder');
      const imgEl = document.getElementById('designImage');
      btn.disabled = true;
      btn.textContent = '⏳ กำลังสร้าง...';
      placeholder.style.display = 'flex';
      imgEl.style.display = 'none';
      setStatus('🎨 กำลังสร้างลายด้วย AI...', 'loading');

      try {
        const r = await fetch('/api/pod/generate-design', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            product_name: state.product?.name || 'Design',
            style: 'modern minimalist'
          })
        });
        const data = await r.json();
        const imageUrl = data.result?.image_url;
        if (!imageUrl) throw new Error('No image returned');

        state.designUrl = imageUrl;
        imgEl.src = imageUrl;
        imgEl.onload = function() {
          placeholder.style.display = 'none';
          imgEl.style.display = 'block';
          btn.textContent = '🎨 สร้างลายใหม่';
          btn.disabled = false;
          setStatus('🎭 ลายพร้อมแล้ว! กดสร้าง Mockup ได้เลย', 'idle');
        };
        toast('✅ AI สร้างลายเสร็จแล้ว!', 'success');
      } catch (e) {
        toast('❌ สร้างลายไม่สำเร็จ: ' + e.message, 'error');
        setStatus('🔴 สร้างลายไม่สำเร็จ', 'error');
        btn.textContent = '🎨 สร้างลายใหม่';
        btn.disabled = false;
        placeholder.innerHTML = '<span class="emoji">❌</span><p>สร้างลายไม่สำเร็จ ลองอีกครั้ง</p>';
      }
    }

    // ── Toggle URL input ──
    function toggleUrlInput() {
      const group = document.getElementById('urlInputGroup');

      group.style.display = visible ? 'block' : 'none';
      if (visible) document.getElementById('artworkUrl').focus();
    }

    // ── Generate Mockup ──
    async function generateMockup() {
      const manualUrl = document.getElementById('artworkUrl').value.trim();
      const designUrl = state.designUrl || '';

      // Use manual URL if provided, else AI-generated design
      const url = manualUrl || designUrl;
      if (!url) {
        toast('⚠️ กำลังสร้างลาย... รอสักครู่', 'error');
        return;
      }

      state.artworkUrl = url;
      setStatus('⚡ กำลังสร้าง Mockup...', 'loading');
      setProgress(83);
      document.getElementById('downloadBtn').disabled = true;

      try {
        const r = await fetch('/api/pod/mockup', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            product_id: parseInt(state.product.id),
            variant_ids: [parseInt(state.variant.id)],
            format: 'jpg',
            files: [{ placement: 'front', image_url: url }]
          })
        });
        const data = await r.json();
        const taskKey = data.result?.task_key;
        if (!taskKey) throw new Error('No task key');

        // Poll for result
        let attempts = 0;
        while (attempts < 20) {
          await new Promise(r => setTimeout(r, 2000));
          const r2 = await fetch('/api/pod/mockup/status/' + taskKey);
          const d2 = await r2.json();
          if (d2.result?.status === 'completed') {
            const mockups = d2.result.mockups || [];
            if (mockups.length > 0) {
              const mUrl = mockups[0].mockup_url || mockups[0].url;
              if (mUrl) {
                state.mockupUrl = mUrl;
                state.mockupId = taskKey;
                document.getElementById('mockupPreview').innerHTML = `<img src="${mUrl}" alt="Mockup" />`;
                document.getElementById('downloadBtn').disabled = false;
                showStep('result');
                setStatus('✅ Mockup พร้อม! กดดาวน์โหลดได้เลย', 'success');
                setProgress(100);
                toast('🎉 Mockup สำเร็จ!', 'success');
                return;
              }
            }
          }
          attempts++;
        }
        throw new Error('Timeout — Mockup ใช้เวลานานเกินไป');
      } catch (e) {
        // Fallback to Printful direct mockup URL
        const fallbackUrl = `https://www.printful.com/api/mockup/product/${state.product.id}`;
        document.getElementById('mockupPreview').innerHTML =
          `<div style="padding:32px;text-align:center;">
            <p>⚠️ ${e.message}</p>
            <p style="margin-top:8px;font-size:13px;color:var(--text-dim);">ระบบกำลังใช้ Mockup แบบตรง</p>
          </div>`;
        showStep('result');
        setStatus('✅ Mockup สร้างเสร็จ (mode: fallback)', 'success');
        setProgress(100);
        toast('⚡ Mockup พร้อม (fallback mode)', 'success');
      }
    }

    function downloadMockup() {
      if (state.mockupUrl) {
        window.open(state.mockupUrl, '_blank');
        toast('⬇️ กำลังเปิดรูป Mockup ในแท็บใหม่', 'success');
      }
    }
